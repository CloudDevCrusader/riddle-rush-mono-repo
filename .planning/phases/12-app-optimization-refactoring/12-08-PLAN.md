Create useAttemptHandler composable extracting attempt logic from game store:

**State to manage:**

**Actions to extract:**

- `recordAttempt(playerId: string, answer: string, isCorrect: boolean)` — Create attempt record
- `validateAttempt(answer: string, categoryId: string)` — Validate answer against category
- `isRoundComplete()` — Check if all players submitted attempts
- `progressToNextRound()` — Advance to next round
- `getRoundAttempts(roundNumber: number)` — Get attempts for specific round
- `getPlayerAttempts(playerId: string)` — Get all attempts for player

**Additional features:**

- `getAttemptStats()` — Calculate attempt statistics (correct %, avg time)
- `clearAttemptHistory()` — Reset for new game
- Track streaks (consecutive correct answers per player)

**Structure:**

```typescript
import { ref, computed } from 'vue'
import type { GameAttempt, Player } from '@riddle-rush/types'
import { useAnswerCheck } from './useAnswerCheck'

export function useAttemptHandler() {
  const currentAttempt = ref<GameAttempt | null>(null)
  const attemptHistory = ref<GameAttempt[]>([])
  const answerCheck = useAnswerCheck()

  async function recordAttempt(
    playerId: string,
    answer: string,
    categoryId: string,
    letter: string
  ): Promise<GameAttempt> {
    const isCorrect = await validateAttempt(answer, categoryId, letter)

    const attempt: GameAttempt = {
      id: crypto.randomUUID(),
      playerId,
      answer,
      isCorrect,
      timestamp: Date.now(),
      roundNumber: getCurrentRound(),
      categoryId,
      letter,
    }

    attemptHistory.value.push(attempt)
    currentAttempt.value = attempt

    return attempt
  }

  async function validateAttempt(
    answer: string,
    categoryId: string,
    letter: string
  ): Promise<boolean> {
    return await answerCheck.checkAnswer(answer, categoryId, letter)
  }

  function isRoundComplete(playerCount: number): boolean {
    const currentRoundAttempts = getRoundAttempts(getCurrentRound())
    return currentRoundAttempts.length >= playerCount
  }

  function progressToNextRound(): void {
    // Clear current attempt
    currentAttempt.value = null
  }

  function getRoundAttempts(roundNumber: number): GameAttempt[] {
    return attemptHistory.value.filter((a) => a.roundNumber === roundNumber)
  }

  // ... more functions

  return {
    currentAttempt: readonly(currentAttempt),
    attemptHistory: readonly(attemptHistory),
    recordAttempt,
    validateAttempt,
    isRoundComplete,
    progressToNextRound,
    getRoundAttempts,
    getPlayerAttempts,
    getAttemptStats,
    clearAttemptHistory,
  }
}
```

```typescript
interface Statistics {
  totalGames: number
  totalScore: number
  averageScore: number
  winRate: number
  playerStats: {
    [playerId: string]: {
      gamesPlayed: number
      totalScore: number
      wins: number
      averageScore: number
    }
  }
}
```

**Structure:**

```typescript
import { ref, computed } from 'vue'
import type { GameSession, Statistics } from '@riddle-rush/types'
import { useIndexedDB } from './useIndexedDB'

export function useStatisticsManager() {
  const statistics = ref<Statistics>({
    totalGames: 0,
    totalScore: 0,
    averageScore: 0,
    winRate: 0,
    playerStats: {},
  })
  const indexedDB = useIndexedDB()

  async function updateStatistics(session: GameSession): Promise<void> {
    // Read from gameHistory store
    const history = await indexedDB.getAllGameHistory()

    // Aggregate statistics
    const totalGames = history.length
    const totalScore = history.reduce((sum, game) => sum + game.totalScore, 0)
    const averageScore = totalGames > 0 ? totalScore / totalGames : 0

    // Per-player stats
    const playerStats = {}
    for (const game of history) {
      for (const player of game.players) {
        if (!playerStats[player.id]) {
          playerStats[player.id] = {
            gamesPlayed: 0,
            totalScore: 0,
            wins: 0,
            averageScore: 0,
          }
        }
        playerStats[player.id].gamesPlayed++
        playerStats[player.id].totalScore += player.score
        if (player.rank === 1) {
          playerStats[player.id].wins++
        }
      }
    }

    // Calculate averages and win rates
    for (const playerId in playerStats) {
      const stats = playerStats[playerId]
      stats.averageScore = stats.totalScore / stats.gamesPlayed
    }

    statistics.value = {
      totalGames,
      totalScore,
      averageScore,
      winRate: 0, // Calculate from player stats
      playerStats,
    }

    // Persist to IndexedDB
    await indexedDB.saveStatistics(statistics.value)
  }

  async function getLeaderboard(limit: number = 10): Promise<any[]> {
    const leaderboard = await indexedDB.getLeaderboard()
    return leaderboard.slice(0, limit)
  }

  // ... more functions

  return {
    statistics: readonly(statistics),
    updateStatistics,
    getPlayerStats,
    getLeaderboard,
    calculateWinRate,
    getRecentGames,
  }
}
```

Create usePersistence composable wrapping useIndexedDB for game store:

**Purpose:** Provide high-level persistence API for game store, hiding IndexedDB complexity.

**Actions to provide:**

- `persist(key: string, value: any)` — Save to IndexedDB store
- `retrieve(key: string)` — Retrieve from IndexedDB store
- `delete(key: string)` — Delete from IndexedDB store
- `persistSession(session: GameSession)` — Save session to gameSession store
- `retrieveSession(id: string)` — Load session from gameSession store
- `persistHistory(session: GameSession)` — Save to gameHistory store
- `persistStatistics(stats: Statistics)` — Save to statistics store

**Structure:**

```typescript
import { useIndexedDB } from './useIndexedDB'
import type { GameSession, Statistics } from '@riddle-rush/types'

export function usePersistence() {
  const indexedDB = useIndexedDB()

  async function persist(key: string, value: any): Promise<void> {
    // Route to appropriate IndexedDB store based on key prefix
    if (key.startsWith('session:')) {
      await indexedDB.saveGameSession(value)
    } else if (key.startsWith('history:')) {
      await indexedDB.saveGameHistory(value)
    } else if (key.startsWith('stats:')) {
      await indexedDB.saveStatistics(value)
    }
  }

  async function retrieve(key: string): Promise<any> {
    if (key.startsWith('session:')) {
      const id = key.replace('session:', '')
      return await indexedDB.getGameSession(id)
    } else if (key.startsWith('history:')) {
      const id = key.replace('history:', '')
      return await indexedDB.getGameHistory(id)
    } else if (key.startsWith('stats:')) {
      return await indexedDB.getStatistics()
    }
    return null
  }

  async function deleteKey(key: string): Promise<void> {
    if (key.startsWith('session:')) {
      const id = key.replace('session:', '')
      await indexedDB.deleteGameSession(id)
    }
  }

  // Convenience methods for game store
  async function persistSession(session: GameSession): Promise<void> {
    await indexedDB.saveGameSession(session)
  }

  async function retrieveSession(id: string): Promise<GameSession | null> {
    return await indexedDB.getGameSession(id)
  }

  async function persistHistory(session: GameSession): Promise<void> {
    await indexedDB.saveGameHistory(session)
  }

  async function persistStatistics(stats: Statistics): Promise<void> {
    await indexedDB.saveStatistics(stats)
  }

  return {
    persist,
    retrieve,
    delete: deleteKey,
    persistSession,
    retrieveSession,
    persistHistory,
    persistStatistics,
  }
}
```

```typescript
export const useGameStore = defineStore('game', () => {
  // Initialize composables
  const categoryManager = useCategoryManager()
  const sessionManager = useSessionManager()
  const scoringEngine = useScoringEngine()
  const attemptHandler = useAttemptHandler()
  const statisticsManager = useStatisticsManager()
  const persistence = usePersistence()

  // Player manager initialized from session
  const playerManager = computed(() => {
    return usePlayerManager(sessionManager.currentSession.value?.players || [])
  })

  // High-level orchestration actions
  async function startNewGame(playerNames: string[], selectedCategories: Category[]) {
    // Orchestrate: fetch categories, create session, initialize players
    await categoryManager.fetchCategories()
    const session = await sessionManager.createSession(playerNames, selectedCategories)
    await persistence.persistSession(session)
  }

  async function submitAnswer(answer: string) {
    // Orchestrate: validate, record attempt, calculate score, update player, persist
    const activePlayer = playerManager.value.getActivePlayer()
    const attempt = await attemptHandler.recordAttempt(activePlayer.id, answer)
    const score = scoringEngine.calculateScore(attempt)
    playerManager.value.updatePlayerScore(activePlayer.id, score)
    await persistence.persistSession(sessionManager.currentSession.value)
  }

  async function endGame() {
    // Orchestrate: determine winners, save history, update statistics
    const winners = scoringEngine.determineWinner(playerManager.value.players.value)
    await persistence.persistHistory(sessionManager.currentSession.value)
    await statisticsManager.updateStatistics(sessionManager.currentSession.value)
  }

  // Expose composable APIs
  return {
    // State
    categories: computed(() => categoryManager.categories.value),
    currentSession: computed(() => sessionManager.currentSession.value),
    players: computed(() => playerManager.value.players.value),
    statistics: computed(() => statisticsManager.statistics.value),

    // Actions (orchestration)
    startNewGame,
    submitAnswer,
    endRound,
    endGame,

    // Direct composable access for components
    categoryManager,
    sessionManager,
    playerManager,
    scoringEngine,
    attemptHandler,
    statisticsManager,
  }
})
```
