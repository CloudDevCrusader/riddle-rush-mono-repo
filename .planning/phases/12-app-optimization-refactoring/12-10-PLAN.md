```bash
check_aws_credentials() {
  log "INFO" "Checking AWS credentials..."
  if ! aws sts get-caller-identity &>/dev/null; then
    log "ERROR" "AWS credentials not configured or invalid"
    log "ERROR" "Run: aws configure"
    exit 1
  fi
  local account_id=$(aws sts get-caller-identity --query Account --output text)
  log "SUCCESS" "AWS credentials valid (Account: $account_id)"
}
```

**2. Git branch check:**

```bash
check_git_branch() {
  local expected_branch=$1
  local current_branch=$(git branch --show-current)
  log "INFO" "Checking git branch..."
  if [ "$current_branch" != "$expected_branch" ]; then
    log "ERROR" "Expected branch '$expected_branch', got '$current_branch'"
    log "ERROR" "Checkout correct branch: git checkout $expected_branch"
    exit 1
  fi
  log "SUCCESS" "On correct branch: $current_branch"
}
```

```bash
check_environment_vars() {
  log "INFO" "Checking environment variables..."
  local required_vars=("AWS_S3_BUCKET" "AWS_CLOUDFRONT_ID" "AWS_REGION")
  for var in "${required_vars[@]}"; do
    if [ -z "${!var}" ]; then
      log "ERROR" "Required environment variable $var not set"
      exit 1
    fi
    log "SUCCESS" "$var set"
  done
}
```

**1. Deployment verification:**

```bash
verify_deployment() {
  local url=$1
  local max_attempts=5
  local attempt=1

  log "INFO" "Verifying deployment at $url..."

  while [ $attempt -le $max_attempts ]; do
    local http_code=$(curl -s -o /dev/null -w "%{http_code}" "$url")
    if [ "$http_code" = "200" ]; then
      log "SUCCESS" "Deployment verified (HTTP 200)"
      return 0
    fi
    log "WARN" "Attempt $attempt/$max_attempts failed (HTTP $http_code), retrying in 10s..."
    sleep 10
    ((attempt++))
  done

  log "ERROR" "Deployment verification failed after $max_attempts attempts"
  return 1
}
```

**2. Backup creation:**

```bash
create_backup() {
  local backup_prefix="backup-$(date +%Y%m%d-%H%M%S)"
  log "INFO" "Creating backup: $backup_prefix"

  # Sync current S3 state to backup prefix
  aws s3 sync "s3://$AWS_S3_BUCKET/" "s3://$AWS_S3_BUCKET/$backup_prefix/" \
    --exclude "$backup_prefix/*" \
    --quiet

  log "SUCCESS" "Backup created at s3://$AWS_S3_BUCKET/$backup_prefix/"
  echo "$backup_prefix"
}
```

**3. Rollback function:**

```bash
rollback_deployment() {
  local backup_prefix=$1

  log "ERROR" "Deployment failed, initiating rollback..."

  # Restore previous S3 version from backup
  aws s3 sync "s3://$AWS_S3_BUCKET/$backup_prefix/" "s3://$AWS_S3_BUCKET/" \
    --delete \
    --exclude "$backup_prefix/*"

  # Invalidate CloudFront cache
  local invalidation_id=$(aws cloudfront create-invalidation \
    --distribution-id "$AWS_CLOUDFRONT_ID" \
    --paths "/*" \
    --query 'Invalidation.Id' \
    --output text)

  log "INFO" "CloudFront invalidation created: $invalidation_id"
  log "SUCCESS" "Rollback complete"
}
```

```bash
#!/bin/bash

# Color codes
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
BLUE='\033[0;34m'
NC='\033[0m' # No Color

log() {
  local level=$1
  shift
  local message="$@"
  local timestamp=$(date +"%Y-%m-%d %H:%M:%S")

  case $level in
    ERROR)
      echo -e "[$timestamp] ${RED}[ERROR]${NC} $message" >&2
      ;;
    SUCCESS)
      echo -e "[$timestamp] ${GREEN}[SUCCESS]${NC} $message"
      ;;
    WARN)
      echo -e "[$timestamp] ${YELLOW}[WARN]${NC} $message"
      ;;
    INFO)
      echo -e "[$timestamp] ${BLUE}[INFO]${NC} $message"
      ;;
    *)
      echo "[$timestamp] [$level] $message"
      ;;
  esac
}

log_section() {
  local title=$1
  echo ""
  echo "=========================================="
  echo "  $title"
  echo "=========================================="
  echo ""
}
```

**2. Post-deployment tasks:**

```bash
post_deployment() {
  local environment=$1
  local version=$(cat package.json | grep '"version"' | head -1 | awk -F: '{ print $2 }' | sed 's/[", ]//g')
  local timestamp=$(date +%Y%m%d-%H%M%S)
  local tag="deploy-$environment-$version-$timestamp"

  log_section "Post-Deployment Tasks"

  # Tag git commit
  log "INFO" "Creating git tag: $tag"
  git tag -a "$tag" -m "Deployment: $environment v$version at $timestamp"
  git push origin "$tag"
  log "SUCCESS" "Git tag created and pushed"

  # Update STATE.md
  log "INFO" "Updating STATE.md..."
  {
    echo ""
    echo "### Deployment: $environment"
    echo "- **Version:** $version"
    echo "- **Timestamp:** $timestamp"
    echo "- **Tag:** $tag"
    echo "- **Branch:** $(git branch --show-current)"
    echo "- **Commit:** $(git rev-parse --short HEAD)"
  } >> .planning/STATE.md

  git add .planning/STATE.md
  git commit -m "chore: record deployment $tag"
  git push
  log "SUCCESS" "STATE.md updated"

  # Summary
  log_section "Deployment Summary"
  log "SUCCESS" "Environment: $environment"
  log "SUCCESS" "Version: $version"
  log "SUCCESS" "Tag: $tag"
  log "SUCCESS" "URL: $(get_deployment_url $environment)"
}

get_deployment_url() {
  local environment=$1
  case $environment in
    production)
      echo "https://riddlerush.de"
      ;;
    development)
      echo "https://dev.riddlerush.de"
      ;;
    staging)
      echo "https://staging.riddlerush.de"
      ;;
  esac
}
```
