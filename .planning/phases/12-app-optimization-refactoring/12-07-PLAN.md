**State to extract:**

- `players: Ref<Player[]>` (from session)

**Actions to extract:**

- `addPlayer(name: string)` — Add new player to session
- `removePlayer(id: string)` — Remove player from session
- `getActivePlayer()` — Get current turn player
- `getPlayerById(id: string)` — Get player by ID
- `updatePlayerScore(id: string, score: number)` — Update player score
- `nextPlayer()` — Advance to next player turn
- `getPlayerRank(playerId: string)` — Get player rank in current standings

**Additional features:**

- `getPlayersCount()` — Count active players
- `getLeadingPlayer()` — Get player with highest score
- `getTiedPlayers()` — Get players with same top score
- `validatePlayerName(name: string)` — Check name is valid (not empty, unique, <20 chars)

**Structure:**

```typescript
import { ref, computed } from 'vue'
import type { Player } from '@riddle-rush/types'

export function usePlayerManager(initialPlayers: Player[] = []) {
  const players = ref<Player[]>(initialPlayers)
  const currentPlayerIndex = ref(0)

  function addPlayer(name: string): Player {
    if (!validatePlayerName(name)) {
      throw new Error('Invalid player name')
    }
    const player: Player = {
      id: crypto.randomUUID(),
      name,
      score: 0,
      isActive: true,
    }
    players.value.push(player)
    return player
  }

  function removePlayer(id: string): boolean {
    const index = players.value.findIndex((p) => p.id === id)
    if (index === -1) return false
    players.value.splice(index, 1)
    return true
  }

  function getActivePlayer(): Player | null {
    if (players.value.length === 0) return null
    return players.value[currentPlayerIndex.value] || null
  }

  function nextPlayer(): void {
    currentPlayerIndex.value = (currentPlayerIndex.value + 1) % players.value.length
  }

  // ... more functions

  return {
    players: readonly(players),
    currentPlayerIndex: readonly(currentPlayerIndex),
    addPlayer,
    removePlayer,
    getActivePlayer,
    getPlayerById,
    updatePlayerScore,
    nextPlayer,
    getPlayerRank,
    getPlayersCount: computed(() => players.value.length),
    getLeadingPlayer,
  }
}
```

Create useScoringEngine composable extracting scoring logic from game store:

**Pure functions (no state):**

- `calculateScore(attempt: GameAttempt)` — Calculate score for correct answer
- `applyBonus(player: Player, bonusType: string)` — Apply bonus points
- `determineWinner(players: Player[])` — Determine winner(s) from final scores
- `getRankSuffix(rank: number)` — Get rank suffix (1st, 2nd, 3rd, 4th...)
- `getScoreDisplay(score: number)` — Format score for display (with + or -)

**Scoring rules:**

- Base score: SCORE_PER_CORRECT_ANSWER (from shared constants)
- Time bonus: +10 points if answered within 10 seconds
- Streak bonus: +5 points per consecutive correct answer
- First answer bonus: +20 points if first to answer in round

**Structure:**

```typescript
import type { GameAttempt, Player } from '@riddle-rush/types'
import { SCORE_PER_CORRECT_ANSWER } from '@riddle-rush/shared'

export function useScoringEngine() {
  function calculateScore(attempt: GameAttempt): number {
    let score = SCORE_PER_CORRECT_ANSWER

    // Time bonus
    if (attempt.timeTaken && attempt.timeTaken < 10000) {
      score += 10
    }

    // Streak bonus (handled by caller passing streak count)
    if (attempt.streak && attempt.streak > 1) {
      score += 5 * (attempt.streak - 1)
    }

    // First answer bonus
    if (attempt.isFirstAnswer) {
      score += 20
    }

    return score
  }

  function determineWinner(players: Player[]): Player[] {
    if (players.length === 0) return []

    const maxScore = Math.max(...players.map((p) => p.score))
    return players.filter((p) => p.score === maxScore)
  }

  function getRankSuffix(rank: number): string {
    const suffixes = ['th', 'st', 'nd', 'rd']
    const v = rank % 100
    return rank + (suffixes[(v - 20) % 10] || suffixes[v] || suffixes[0])
  }

  function getScoreDisplay(score: number): string {
    if (score === 0) return '0'
    return score > 0 ? `+${score}` : `${score}`
  }

  return {
    calculateScore,
    applyBonus: (player: Player, bonusPoints: number) => player.score + bonusPoints,
    determineWinner,
    getRankSuffix,
    getScoreDisplay,
  }
}
```
