**Structure:**

```typescript
import { ref, computed } from 'vue'
import type { GameSession, Player, Category } from '@riddle-rush/types'
import { useIndexedDB } from './useIndexedDB'

export function useSessionManager() {
  const currentSession = ref<GameSession | null>(null)
  const indexedDB = useIndexedDB()

  async function createSession(players: Player[], categories: Category[]): Promise<GameSession> {
    const session: GameSession = {
      id: crypto.randomUUID(),
      startTime: Date.now(),
      players,
      categories,
      currentRound: 1,
      maxRounds: 5,
      status: 'active',
    }
    currentSession.value = session
    await saveSession(session)
    return session
  }

  async function loadSessionById(id: string): Promise<GameSession | null> {
    const session = await indexedDB.getGameSession(id)
    if (session) {
      currentSession.value = session
    }
    return session
  }

  async function saveSession(session: GameSession): Promise<void> {
    await indexedDB.saveGameSession(session)
  }

  // ... more functions

  return {
    currentSession: readonly(currentSession),
    createSession,
    loadSessionById,
    saveSession,
    cloneSessionForHistory,
    clearSession,
  }
}
```

Use crypto.randomUUID() for session IDs.
