       ```typescript
       // In apps/game/stores/game.ts or types/game.ts
       export type GameModeState =
         | 'menu'           // Main menu
         | 'players'        // Player setup
         | 'round-start'    // Fortune wheel / round start
         | 'playing'        // Active game round
         | 'scoring'        // Score assignment phase
         | 'results'        // Results/leaderboard
         | 'paused'         // Game paused overlay
         | 'settings'       // Settings screen
         | 'language'       // Language selection
         | 'credits'        // Credits screen

       // Add state metadata:
       export interface GameModeStateMeta {
         current: GameModeState
         previous?: GameModeState
         canPause: boolean     // true for 'playing' only
         canResume: boolean    // true for 'paused' only
         canQuit: boolean      // true for all non-menu states
       }
       ```

    2. Add game mode to unified store:
       ```typescript
       // In stores/game.ts:
       const gameMode = ref<GameModeState>('menu')
       const gameModeMeta = computed<GameModeStateMeta>(() => {
         return {
           current: gameMode.value,
           previous: previousGameMode.value,
           canPause: gameMode.value === 'playing',
           canResume: gameMode.value === 'paused',
           canQuit: gameMode.value !== 'menu'
         }
       })

       // Transition function:
       function setGameMode(newMode: GameModeState, reason?: string): void {
         previousGameMode.value = gameMode.value
         gameMode.value = newMode
         logger?.info(`Game mode: ${gameMode.value} -> ${newMode}`, { reason })
       }

       // Expose from store:
       return {
         gameMode: readonly(gameMode),
         gameModeMeta: gameModeMeta,
         setGameMode
       }
       ```

    3. Update or create useGameState composable:
       ```typescript
       // In composables/useGameState.ts:
       export function useGameState() {
         const gameMode = computed(() => useGameStore().gameMode)
         const gameModeMeta = computed(() => useGameStore().gameModeMeta)
         const setGameMode = (mode: GameModeState, reason?: string) =>
           useGameStore().setGameMode(mode, reason)

         return {
           gameMode,
           gameModeMeta,
           setGameMode,
           isMenu: computed(() => gameMode.value === 'menu'),
           isPlaying: computed(() => gameMode.value === 'playing'),
           isPaused: computed(() => gameMode.value === 'paused'),
           isScoring: computed(() => gameMode.value === 'scoring'),
           isResults: computed(() => gameMode.value === 'results'),
         }
       }
       ```

    4. Remove duplicate game mode state definitions:
       - Search codebase for other gameMode variables
       - Replace with calls to setGameMode()
       - Ensure single source of truth
